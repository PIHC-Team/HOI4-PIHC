PIHC_update_current_month = {
	add_to_variable = {
        global.PIHC_var_current_month = 1
    }
	modulo_variable = {
		global.PIHC_var_current_month = 12
	}
	if = {
		limit = {
			check_variable = {
				global.PIHC_var_current_month = 0
			}
		}
		set_variable = {
			global.PIHC_var_current_month = 12
		}
	}

	PIHC_update_monthly_global_variables = yes
}

PIHC_change_ERA_COLD_level = {#更新政策等级
	if = {
        limit = {
            has_idea = IDEA_ERA_COLD_1_COAL
            check_variable = {
                resource@coal < 0
            }
        }
        add_ideas = IDEA_ERA_COLD_2_WOOD
    }
    else_if = {
        limit = {
            has_idea = IDEA_ERA_COLD_2_WOOD
            check_variable = {
                resource@coal < 0
                resource@logs < 0
            }
        }
        add_ideas = IDEA_ERA_COLD_3_SUBSIDY
    }
}

PIHC_update_state_indoor_temperature = {#更新室内温度
	set_variable = {
		THIS.PIHC_var_state_indoor_temperature = THIS.PIHC_var_state_outdoor_temperature
	}

	#地块内供热塔
	set_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = THIS.building_level@generator_complex
	}
	multiply_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = 15
	}
	add_to_variable = {
		THIS.PIHC_var_state_indoor_temperature = THIS.PIHC_var_state_temperature_factor_gc
	}

	#相邻地块供热塔
	set_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = 0
	}
	every_neighbor_state = {
		if = {
			limit = {
				generator_complex > 0
			}
			add_to_variable = {
				PREV.PIHC_var_state_temperature_factor_gc = 1
			}
		}
	}
	multiply_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = 5
	}
	add_to_variable = {
		THIS.PIHC_var_state_indoor_temperature = THIS.PIHC_var_state_temperature_factor_gc
	}

	#御寒科技
	set_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = 3
	}
	multiply_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = THIS.PIHC_var_country_tec_COLD_level
	}
	add_to_variable = {
		THIS.PIHC_var_state_indoor_temperature = THIS.PIHC_var_state_temperature_factor_gc
	}

	#供暖政策
	set_variable = {

	}
	add_to_variable = {
		THIS.PIHC_var_state_temperature_factor_gc = THIS.PIHC_var_country_tec_COLD_level

	}
	add_to_variable = {
