sabotage_infrastructure = {
	available = {
		non_damaged_building_level = {
			building = infrastructure
			level > 0
		}
	}
	weight = {
		base = 1
		modifier = {
			has_variable = targeted_sabotage_infrastructure
			check_variable = {
				targeted_sabotage_infrastructure > 0
			}
			add = 4
		}
	}
	effect = {
		if = {
			limit = {
				has_variable = targeted_sabotage_infrastructure
				check_variable = {
					targeted_sabotage_infrastructure > 0
				}
			}
			damage_building = {
				type = infrastructure
				damage = 1.0
			}
			subtract_from_variable = {
				var = targeted_sabotage_infrastructure
				value = 1
			}
		}
		else = {
			damage_building = {
				type = infrastructure
				damage = 0.5
			}
		}
	}
	alert_text = building_is_sabotaged
}
