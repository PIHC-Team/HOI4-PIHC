C15_PARLIAMENT_MODIFY_SEAT = {
	subtract_from_temp_variable = {
		temp_member = 1
	}
	set_temp_variable = {
		temp = C15.parliament_seat_array^temp_member
	}
	add_to_temp_variable = {
		temp = temp_seats
	}
	if = {
		limit = {
			check_variable = {
				temp > C15.parliament_main_array^num
			}
		}
		subtract_from_temp_variable = {
			temp = C15.parliament_main_array^num
		}
		subtract_from_temp_variable = {
			temp_seats = temp
		}
	}
	else_if = {
		limit = {
			check_variable = {
				temp < 0
			}
		}
		multiply_variable = {
			temp = -1
		}
		subtract_from_temp_variable = {
			temp_seats = temp
		}
	}
	for_loop_effect = {
		start = 0
		end = C15.parliament_seat_array^num
		add_to_temp_array = {
			temp_array = v
		}
	}
	remove_from_temp_array = {
		array = temp_array
		value = temp_member
	}
	add_to_variable = {
		C15.parliament_seat_array^temp_member = temp_seats
	}
	if = {
		limit = {
			check_variable = {
				temp_seats > 0
			}
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					temp_seats > 0
				}
			}
			for_loop_effect = {
				start = 0
				end = temp_array^num
				if = {
					limit = {
						check_variable = {
							C15.parliament_seat_array^temp_array^v > 0
						}
						check_variable = {
							temp_seats > 0
						}
					}
					subtract_from_variable = {
						C15.parliament_seat_array^temp_array^v = 1
					}
					subtract_from_temp_variable = {
						temp_seats = 1
					}
				}
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				temp_seats < 0
			}
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					temp_seats < 0
				}
			}
			for_loop_effect = {
				start = 0
				end = temp_array^num
				if = {
					limit = {
						check_variable = {
							C15.parliament_seat_array^temp_array^v < C15.parliament_main_array^num
						}
						check_variable = {
							temp_seats < 0
						}
					}
					add_to_variable = {
						C15.parliament_seat_array^temp_array^v = 1
					}
					add_to_temp_variable = {
						temp_seats = 1
					}
				}
			}
		}
	}
	C15_PARLIAMENT_GUI_RELOAD = yes
}
