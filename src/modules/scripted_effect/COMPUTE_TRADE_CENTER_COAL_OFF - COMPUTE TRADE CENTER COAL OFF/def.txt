COMPUTE_TRADE_CENTER_COAL_OFF = {
	if = {
		limit = {
			OR = {
				has_country_flag = PIHC_COUNTRY_FLAG_AUTO_TRADE
				has_country_flag = PIHC_COUNTRY_FLAG_AUTO_TRADE_COAL_SELL
			}
		}
		set_variable = {
			var = VAR_TRADE_CENTER_COAL_OFF
			value = resource@coal
		}
		subtract_from_variable = {
			var = VAR_TRADE_CENTER_COAL_OFF
			value = VAR_TRADE_CENTER_COAL_PRESERVE
		}
		clamp_variable = {
			var = VAR_TRADE_CENTER_COAL_OFF
			min = 0
		}
	}
	else = {
		set_variable = {
			var = VAR_TRADE_CENTER_COAL_OFF
			value = 0
		}
	}
}
