COMPUTE_NUM_ALUMINIUM_MILLS = {
	set_variable = {
		var = THIS.VAR_NUM_ALUMINIUM_MILLS_OWNED
		value = 0
	}
	every_controlled_state = {
		COMPUTE_NUM_ALUMINIUM_MILLS_IN_STATE = yes
		add_to_variable = {
			var = PREV.VAR_NUM_ALUMINIUM_MILLS_OWNED
			value = THIS.VAR_NUM_ALUMINIUM_MILLS_IN_CURRENT_STATE
		}
	}
}
