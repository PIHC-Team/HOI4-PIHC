	custom_effect_tooltip = PIHC_DOWNGRADE_TECH_TOOLTIP

			has_idea = IDEA_ERA_TECH_2_CHEER
		}
		swap_ideas = {
			add_idea = IDEA_ERA_TECH_3_NEUTRAL
			remove_idea = IDEA_ERA_TECH_2_CHEER
		}
	}
	else_if = {
		limit = {
			has_idea = IDEA_ERA_TECH_3_NEUTRAL
		}
		swap_ideas = {
			add_idea = IDEA_ERA_TECH_4_FEAR
			remove_idea = IDEA_ERA_TECH_3_NEUTRAL
		}
	}
	else_if = {
		limit = {
			has_idea = IDEA_ERA_TECH_4_FEAR
		}
		swap_ideas = {
			add_idea = IDEA_ERA_TECH_5_STAGNATE
			remove_idea = IDEA_ERA_TECH_4_FEAR
		}
	}
	else_if = {
		limit = {
			has_idea = IDEA_ERA_TECH_5_STAGNATE
		}
		swap_ideas = {
			add_idea = IDEA_ERA_TECH_6_DECLINE
			remove_idea = IDEA_ERA_TECH_5_STAGNATE
		}
	}
}
