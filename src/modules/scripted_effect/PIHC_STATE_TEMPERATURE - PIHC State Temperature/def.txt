PIHC_update_current_month = {
	PIHC_sync_temperature_calendar = yes
	PIHC_update_monthly_global_variables = yes
}

# HoI4's global.num_days uses a 365-day calendar (verified against native dates).
# Derive the season from the engine date, never from the number of callbacks.
PIHC_sync_temperature_calendar = {
	set_variable = { global.PIHC_temperature_day_of_year = global.num_days }
	modulo_variable = { global.PIHC_temperature_day_of_year = 365 }
	if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 31 } }
		set_variable = { global.PIHC_var_current_month = 1 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 59 } }
		set_variable = { global.PIHC_var_current_month = 2 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 90 } }
		set_variable = { global.PIHC_var_current_month = 3 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 120 } }
		set_variable = { global.PIHC_var_current_month = 4 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 151 } }
		set_variable = { global.PIHC_var_current_month = 5 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 181 } }
		set_variable = { global.PIHC_var_current_month = 6 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 212 } }
		set_variable = { global.PIHC_var_current_month = 7 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 243 } }
		set_variable = { global.PIHC_var_current_month = 8 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 273 } }
		set_variable = { global.PIHC_var_current_month = 9 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 304 } }
		set_variable = { global.PIHC_var_current_month = 10 }
	}
	else_if = {
		limit = { check_variable = { global.PIHC_temperature_day_of_year < 334 } }
		set_variable = { global.PIHC_var_current_month = 11 }
	}
	else = { set_variable = { global.PIHC_var_current_month = 12 } }
	# A unique year/month key also guards staggered country callbacks.
	set_variable = { global.PIHC_temperature_calendar_key = global.num_days }
	subtract_from_variable = { global.PIHC_temperature_calendar_key = global.PIHC_temperature_day_of_year }
	add_to_variable = { global.PIHC_temperature_calendar_key = global.PIHC_var_current_month }
}

PIHC_update_monthly_global_variables = {
	# No weekly work. One seasonal lookup and one temperature evaluation per state per month.
	PIHC_prepare_temperature_season = yes
	set_variable = { global.PIHC_var_state_temperature_factor_threat = global.threat }
	multiply_variable = { global.PIHC_var_state_temperature_factor_threat = 60 }
	every_country = { PIHC_prepare_country_temperature = yes }
	# Scatter neighbor heat from towers only; no per-state adjacency traversal.
	every_state = { set_variable = { PIHC_temperature_neighbor_heat = 0 } }
	every_state = {
		limit = { generator_complex > 0 }
		every_neighbor_state = { add_to_variable = { PIHC_temperature_neighbor_heat = 5 } }
	}
	every_country = {
		set_variable = { THIS.PIHC_var_state_subsidies_cost_coal = 0 }
		set_variable = { THIS.PIHC_var_state_subsidies_cost_logs = 0 }
		set_variable = {
			THIS.PIHC_var_country_temperature_weighted_comfort_sum = 0
		}
		set_variable = {
			THIS.PIHC_var_country_temperature_population_sum = 0
		}
		every_controlled_state = {
			set_variable = { THIS.PIHC_var_state_temperature_factor_threat_con = PREV.PIHC_var_state_temperature_factor_threat_con }
			set_variable = { THIS.PIHC_var_state_controller_outdoor_temperature_influence = PREV.PIHC_var_state_controller_outdoor_temperature_influence }
			set_variable = { THIS.PIHC_var_country_ERA_COLD_level = PREV.PIHC_var_country_ERA_COLD_level }
			set_variable = { THIS.PIHC_var_country_tec_COLD_level = PREV.PIHC_var_country_tec_COLD_level }
			PIHC_update_state_outdoor_temperature = yes
			add_to_variable = { PREV.PIHC_var_state_subsidies_cost_coal = THIS.coal_subsidies }
			add_to_variable = { PREV.PIHC_var_state_subsidies_cost_logs = THIS.logs_subsidies }
			set_variable = {
				THIS.PIHC_var_state_temperature_population_weight = THIS.state_population_k
			}
			divide_variable = { THIS.PIHC_var_state_temperature_population_weight = 1000 }
			set_variable = {
				THIS.PIHC_var_state_temperature_weighted_comfort = THIS.PIHC_var_state_temperature_population_weight
			}
			multiply_variable = {
				THIS.PIHC_var_state_temperature_weighted_comfort = THIS.PIHC_var_state_indoor_temperature_comfort_score
			}
			add_to_variable = {
				PREV.PIHC_var_country_temperature_population_sum = THIS.PIHC_var_state_temperature_population_weight
			}
			add_to_variable = {
				PREV.PIHC_var_country_temperature_weighted_comfort_sum = THIS.PIHC_var_state_temperature_weighted_comfort
			}
		}
		PIHC_update_average_temperature_buff = yes
	}
	set_global_flag = PIHC_temperature_initialized
	add_to_variable = { global.PIHC_temperature_revision = 1 }
}

PIHC_change_ERA_COLD_level = {
	if = {
		limit = {
			has_idea = IDEA_ERA_COLD_1_COAL
			check_variable = {
				resource@coal < 0
			}
		}
		add_ideas = IDEA_ERA_COLD_2_WOOD
	}
	else_if = {
		limit = {
			has_idea = IDEA_ERA_COLD_2_WOOD
			check_variable = {
				resource@logs < 0
			}
		}
		add_ideas = IDEA_ERA_COLD_3_SUBSIDY
	}
}

PIHC_update_state_outdoor_temperature = {
	set_variable_to_random = {
		var = THIS.PIHC_var_state_outdoor_temperature
		min = global.PIHC_temperature_season_min
		max = global.PIHC_temperature_season_max
		integer = yes
	}

	set_variable = {
		THIS.PIHC_var_state_terrain_temperature_correction = THIS.VAR_OUTDOOR_TEMPERATURE_INFLUENCE_MAX
	}
	subtract_from_variable = {
		THIS.PIHC_var_state_terrain_temperature_correction = THIS.VAR_OUTDOOR_TEMPERATURE_INFLUENCE_MIN
	}
	set_variable_to_random = {
		var = THIS.PIHC_var_state_terrain_temperature_roll
		min = 0
		max = 1
	}
	multiply_variable = {
		THIS.PIHC_var_state_terrain_temperature_correction = THIS.PIHC_var_state_terrain_temperature_roll
	}
	add_to_variable = {
		THIS.PIHC_var_state_terrain_temperature_correction = THIS.VAR_OUTDOOR_TEMPERATURE_INFLUENCE_MIN
	}
	round_variable = THIS.PIHC_var_state_terrain_temperature_correction
	add_to_variable = {
		THIS.PIHC_var_state_outdoor_temperature = THIS.PIHC_var_state_terrain_temperature_correction
	}
	add_to_variable = {
		THIS.PIHC_var_state_outdoor_temperature = THIS.PIHC_var_state_controller_outdoor_temperature_influence
	}
	subtract_from_variable = {
		THIS.PIHC_var_state_outdoor_temperature = global.PIHC_var_state_temperature_factor_threat
	}
	subtract_from_variable = {
		THIS.PIHC_var_state_outdoor_temperature = THIS.PIHC_var_state_temperature_factor_threat_con
	}
	clamp_variable = {
		var = THIS.PIHC_var_state_outdoor_temperature
		min = -50
		max = 50
	}

	PIHC_update_state_outdoor_temperature_frame = yes
	PIHC_update_state_indoor_temperature = yes
}

PIHC_update_state_outdoor_temperature_frame = {
	set_variable = { THIS.PIHC_var_state_outdoor_temperature_display = THIS.PIHC_var_state_outdoor_temperature }
	clamp_variable = { var = THIS.PIHC_var_state_outdoor_temperature_display min = -39 max = 42 }
	# Existing mercury art has its top at y=39. Its clipped container starts at y=40.
	# Child coordinates/clipping do not inherit the parent 1.28 scale.
	# y = 1.28 * (129 - 4*T); top moves along the scaled tube.
	set_variable = { THIS.PIHC_temperature_mercury_y = THIS.PIHC_var_state_outdoor_temperature_display }
	multiply_variable = { THIS.PIHC_temperature_mercury_y = -5.12 }
	add_to_variable = { THIS.PIHC_temperature_mercury_y = 165.12 }
}

			limit = { check_variable = { PREV.PIHC_var_country_temperature_population_sum > 0 } }
			set_variable = { THIS.PIHC_var_state_temperature_population_share_percent = THIS.PIHC_var_state_temperature_population_weight }
			divide_variable = { THIS.PIHC_var_state_temperature_population_share_percent = PREV.PIHC_var_country_temperature_population_sum }
			multiply_variable = { THIS.PIHC_var_state_temperature_population_share_percent = 100 }
		}

		set_variable = {
			THIS.PIHC_var_state_controller_temperature_comfort_score = PREV.PIHC_var_country_temperature_comfort_score
		}
		set_variable = {
			THIS.PIHC_var_state_controller_monthly_population = PREV.PIHC_var_country_monthly_population
		}
		set_variable = {
			THIS.PIHC_var_state_controller_winter_attrition_factor = PREV.PIHC_var_country_winter_attrition_factor
		}
		set_variable = {
			THIS.PIHC_var_state_controller_cold_gain_factor = PREV.PIHC_var_country_cold_gain_factor
		}
		set_variable = {
			THIS.PIHC_var_state_controller_production_speed_buildings_factor = PREV.PIHC_var_country_production_speed_buildings_factor
		}
	}

	force_update_dynamic_modifier = yes

PIHC_prepare_country_temperature = {

		set_variable = { THIS.PIHC_var_country_ERA_COLD_level = 0 }
		set_variable = { THIS.PIHC_var_country_tec_COLD_level = 0 }
		set_variable = { THIS.PIHC_var_state_temperature_factor_threat_con = THIS.has_added_tension_amount }
		multiply_variable = { THIS.PIHC_var_state_temperature_factor_threat_con = 0.3 }
		set_variable = { THIS.PIHC_var_state_controller_outdoor_temperature_influence = THIS.VAR_OUTDOOR_TEMPERATURE_INFLUENCE }
		# Preserve the original first-month regional cold snap, without a second state update.
		if = {
			limit = { NOT = { has_global_flag = PIHC_temperature_initialized } check_variable = { global.num_days = 366064 } OR = { tag = C15 tag = C17 tag = C27 } }
			add_to_variable = { THIS.PIHC_var_state_controller_outdoor_temperature_influence = -8 }
		}

		set_variable = {
			THIS.PIHC_var_country_resource_cost_coal = THIS.num_controlled_states
		}
		set_variable = {
			THIS.PIHC_var_country_resource_cost_logs = THIS.num_controlled_states
		}
		divide_variable = {
			THIS.PIHC_var_country_resource_cost_coal = 3
		}
		divide_variable = {
			THIS.PIHC_var_country_resource_cost_logs = 2
		}
		round_variable = THIS.PIHC_var_country_resource_cost_coal
		round_variable = THIS.PIHC_var_country_resource_cost_logs

		set_variable = {
			THIS.PIHC_var_country_resource_cost_civ = THIS.PIHC_var_country_resource_cost_logs
		}
		divide_variable = {
			THIS.PIHC_var_country_resource_cost_civ = 4
		}
		round_variable = THIS.PIHC_var_country_resource_cost_civ

		set_variable = {
			THIS.PIHC_var_country_modifier_cost_coal = 0
		}
		set_variable = {
			THIS.PIHC_var_country_modifier_cost_logs = 0
		}
		set_variable = {
			THIS.PIHC_var_country_modifier_cost_civ = 0
		}

		PIHC_change_ERA_COLD_level = yes

		if = {
			limit = {
				has_idea = IDEA_ERA_COLD_1_COAL
			}
			set_variable = { THIS.PIHC_var_country_modifier_cost_coal = THIS.PIHC_var_country_resource_cost_coal }
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_logs = 0
			}
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_civ = 0
			}
			set_variable = { THIS.PIHC_var_country_ERA_COLD_level = 3 }
		}
		else_if = {
			limit = {
				has_idea = IDEA_ERA_COLD_2_WOOD
			}
			set_variable = { THIS.PIHC_var_country_modifier_cost_logs = THIS.PIHC_var_country_resource_cost_logs }
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_coal = 0
			}
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_civ = 0
			}
			set_variable = { THIS.PIHC_var_country_ERA_COLD_level = 2 }
		}
		else_if = {
			limit = {
				has_idea = IDEA_ERA_COLD_3_SUBSIDY
			}
			set_variable = { THIS.PIHC_var_country_modifier_cost_civ = THIS.PIHC_var_country_resource_cost_civ }
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_coal = 0
			}
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_logs = 0
			}
			set_variable = { THIS.PIHC_var_country_ERA_COLD_level = 1 }
		}
		else_if = {
			limit = {
				has_idea = IDEA_ERA_COLD_5_BURN
			}
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_coal = 0
			}
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_logs = 0
			}
			set_variable = {
				THIS.PIHC_var_country_modifier_cost_civ = 0
			}
			set_variable = { THIS.PIHC_var_country_ERA_COLD_level = -1 }
		}

		if = {
			limit = {
				has_tech = TECHNOLOGY_SUPPORT_COLD_IV
			}
			set_variable = { THIS.PIHC_var_country_tec_COLD_level = 4 }
		}
		else_if = {
			limit = {
				has_tech = TECHNOLOGY_SUPPORT_COLD_III
			}
			set_variable = { THIS.PIHC_var_country_tec_COLD_level = 3 }
		}
		else_if = {
			limit = {
				has_tech = TECHNOLOGY_SUPPORT_COLD_II
			}
			set_variable = { THIS.PIHC_var_country_tec_COLD_level = 2 }
		}
		else_if = {
			limit = {
				has_tech = TECHNOLOGY_SUPPORT_COLD_I
			}
			set_variable = { THIS.PIHC_var_country_tec_COLD_level = 1 }
		}

}
