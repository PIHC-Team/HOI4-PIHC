COMPUTE_NUM_FINANCIAL_CENTERS = {
	set_variable = {
		var = THIS.VAR_NUM_FINANCIAL_CENTERS_OWNED
		value = 0
	}
	every_controlled_state = {
		COMPUTE_NUM_FINANCIAL_CENTERS_IN_STATE = yes
		add_to_variable = {
			var = PREV.VAR_NUM_FINANCIAL_CENTERS_OWNED
			value = THIS.VAR_NUM_FINANCIAL_CENTERS_IN_CURRENT_STATE
		}
	}
}
